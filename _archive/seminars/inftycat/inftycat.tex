\begin{definition*}
A \emph{model category} is a bicomplete category with a \emph{model structure}.
\end{definition*}

\begin{definition*}[C$^*$-category]
A C$^*$-category is a category $A$ which is a $\C$-linear category with conjugate-linear involutions $*:A(x,y)\to A(y,x)$ and complete norms $\|\cdot\|:A(x,y)\to[0,\infty)$ on each hom-set $A(x,y)$ such that $\|fg\|\le\|f\|\|g\|$ and $\|f^*f\|=\|f\|^2$ and $f^*f\ge0$ are satisfied.
\end{definition*}
\begin{example*}
A C$^*$-algebra is a C$^*$-category with a single object.
\end{example*}
\begin{example*}
The category $\mathrm{Hilb}$ defined such that objects are separable Hilbert spaces and morphisms are bounded linear operators is a C$^*$-category. 
\end{example*}
\begin{definition*}
The category $\mathrm{C^*Cat}$ is the category of C$^*$-categories whose morphisms are given by $*$-functors.
\end{definition*}

\begin{definition*}
Two C$^*$-categories $A$ and $B$ are called \emph{unitary equivalent} if there are $*$-functors $F:A\to B$ and $G:B\to A$ such that $GF\simeq\id_A$ and $FG\simeq\id_B$.
Here, $GF\simeq\id_A$ means that for each morphism $f:a\to a'$ in $A$, we have unitaries $u_a:GF(a)\to a$ and $u_{a'}:GF(a')\to a'$ such that $fu_a=u_{a'}GF(f)$.
The unitarity of $u_a$ means that $u_a^*u_a=\id_{GF(a)}$ and $u_au_a^*=\id_a$.
\end{definition*}
\begin{definition*}
We define a model structure on $\mathrm{C^*Cat}$ as follows:
\begin{itemize}
\item weak equivalences are unitary equivalences,
\item cofibrations are $*$-functors injective on objects,
\item fibrations are $*$-functors such that whenever there is a unitary (equivalence) $u:b\sim Fa'$ for some $a'\in A$ we have $a\in A$ and unitary $v:a\to a'$ such that $Fv=u$, and hence $Fa=b$.
\end{itemize}
\end{definition*}

\begin{enumerate}[(i)]
\item an object $i$ is given by a pair $(A_i,f_i)$ consisting of $A_i\in\cA$ and $f_i:A_i\to X$,
\item a morphism $i'\to i$ is given by a morphism $f_{i'i}:A_{i'}\to A_i$ in $\cA$ satisfying $f_if_{i'i}=f_{i'}$.
\end{enumerate}

Note that for every $f:A\to X$ with $A\in\cA$, there is unique $i\in\cI$ such that $f=f_i$, and for every $h:A'\to A$ and $f:A\to X$ with $A,A'\in\cA$, there are unique $i,i'\in\cI$ such that $f=f_i$ and $f_ih=f_{i'}$.
With these $i,i'$, the last condition implies that $h$ induces a morphism $i'\to i$.

We claim $X=\colim_iA_i$ in $\hat\cA$.
Choose a cocone $g_i:A_i\to Y$ in $\hat\cA$ for the diagram $A_i$.

\[Y(h)(g_A(f))=Y(h)(g_i)=g_ih=g_{i'}=g_{A_{i'}}(f_{i'})=g_{A_{i'}}(f_ih)=g_{A'}(X(h)(f)),\qquad f\in\Hom(A,X),\]

\[(gf_i)_{A'}(h)=g_{A'}((f_i)_{A'}(h))=g_{A'}(f_ih)=g_{A'}(f_{i'})=g_{i'}=g_ih=(g_i)_{A'}(h),\qquad h\in\Hom(A',A_i),\]

\end{proof}

\begin{definition*}[Coproducts and products]
Let $X$ and $Y$ be simplicial sets.
Since the category of simplicial sets is just another name of the category of presheaves on the simplex category, many categorical constructions that can be done in the category of sets are valid.
For example, the initial and terminal simplicial sets are $\varnothing$ and $*:=\Delta^0$, and the coproduct and product are given by
\[(X\sqcup Y)_n:=X_n\sqcup Y_n,\qquad(X\times Y)_n:=X_n\times Y_n.\]
\end{definition*}

\begin{definition*}[Mapping complexes]
Let $X$ and $Y$ be simplicial sets.
The \emph{mapping complex} $\Map(X,Y)$ is a simplicial set defined by
\[\Map(X,Y)_n:=\Hom_{\Set_\Delta}(\Delta^n\times X,Y),\]
and the face maps and degeneracy maps are canonically defined by
\begin{align*}
d_i^nf_n&:=f_n\circ(\delta_n^i\times\id)\in\Hom_{\Set_\Delta}(\Delta^{n-1}\times X,Y),\\
s_i^nf_n&:=f_n\circ(\sigma_n^i\times\id)\in\Hom_{\Set_\Delta}(\Delta^{n+1}\times X,Y),
\end{align*}
for $f_n\in\Hom_{\Set_\Delta}(\Delta^n\times X,Y)$ and $0\le i\le n$.
The simplicial identities for $d$ and $s$ follow from the cosimplicial identities for $\delta$ and $\sigma$.
\end{definition*}

\subsection{Homotopy theory}


\begin{definition*}[Connected components]
A \emph{summand} of a simplicial set is a simplicial subset whose degree-wise complement is also a simplicial subset.
A simplicial set is called \emph{connected} if it is non-empty and there is no non-trivial summand.
A \emph{connected component} of a simplicial set is a connected summand.
The set of components of a simplicial set $X$ is denoted by $\pi_0(X)$.
\end{definition*}

\begin{proposition*}
Let $X$ be a simplicial set.

\begin{parts}
\item
$\pi_0$ is given by the left Kan extension of the constant functor $\Delta\to\Set$ along the Yoneda embedding.
\item
$\pi_0$ is exhibited as the coequalizer of the face maps $d_0$ and $d_1$ for each simplicial set.
\end{parts}
\end{proposition*}
\begin{proof}
(a)
As a lemma we first show that for a simplicial map $f:X\to Y$ from connected $X$ there is a unique connected component of $Y$ containing $f(X)$.
Let $Y'$ be the smallest summand containing $f(X)$, which exists by taking intersections.
Since $X$ and $Y'$ should be non-empty, it is enough to show $Y'$ is connected.
If $Y''$ is a summand of $Y'$, then $f^{-1}(Y'')$ is a summand of $X$, so $Y''=Y'$ or $Y''=\varnothing$, and this says that $Y'$ is connected.

Now we prove the statement.
